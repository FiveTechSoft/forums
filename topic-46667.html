Para crear un archivo .INI con atributo oculto en Windows desde Harbour/xHarbour/FiveWin, usa la API de Windows o la función de Harbour:

Opción 1: hb_fAttrSet() (Harbour estándar)

#include "fileio.ch"

LOCAL cFile := "config.ini"
LOCAL nHandle

// Crear/abrir el archivo
nHandle := FCreate( cFile, FC_NORMAL )
IF nHandle != F_ERROR
   FClose( nHandle )
   // Establecer atributo oculto (FA_HIDDEN = 2)
   hb_fAttrSet( cFile, HB_FA_HIDDEN )
ENDIF

Opción 2: SetFileAttributes() vía HB_FATTRSET / hb_vfAttrSet()

#include "hbwin.ch"

LOCAL cFile := "config.ini"

hb_vfAttrSet( cFile, FILE_ATTRIBUTE_HIDDEN )  // 0x02

Opción 3: Directo con Win32 API (FiveWin)

#include "fivewin.ch"

LOCAL cFile := "config.ini"

SetFileAttributes( cFile, FILE_ATTRIBUTE_HIDDEN )

Notas:
- FILE_ATTRIBUTE_HIDDEN = 0x02 (constante en fivewin.ch / hbwin.ch).
- Si el archivo ya existe, basta llamar a la función de atributos; no hace falta re-crearlo.
- Para oculto + sistema (menos visible en Explorer): FILE_ATTRIBUTE_HIDDEN + FILE_ATTRIBUTE_SYSTEM (0x06).

#include "FiveWin.ch"
#include "fileio.ch"

FUNCTION Main()

   LOCAL oIni, Nome_Arq, nRegistro
   LOCAL cFileIni := ".\ARMANDO.ini"

   // 1. Cria o arquivo INI e grava os dados normalmente
   INI oIni FILE cFileIni

  SET SECTION "Usuario" ENTRY "Nombre" TO "Armando SOI" OF oIni
  SET SECTION "Usuario" ENTRY "Activo" TO "S"           OF oIni

   ENDINI

   // Atributos: +h - invisible -h visible.
   IF File( cFileIni )

  // Cria o Arquivo de Lote *.Bat no Diret¢rio, Com as Diretrizes.
  IF .NOT. FILE( "INV.BAT" )

     NOME_ARQ := FCREATE("INV.BAT")

     NREGISTRO := "@ECHO OFF"               ;
                  + CRLF        +           ;
                  "CLS"         +           ;
                  + CRLF + CRLF +           ;
                  "ATTRIB +h ARMANDO.INI" + ;
                  + CRLF + CRLF +           ;
                  "CLS"         +           ;
                  + CRLF        +           ;
                  "EXIT"

     FWRITE( NOME_ARQ, NREGISTRO )
     FCLOSE( NOME_ARQ )

  ENDIF

  WinExec( "INV.BAT", 0 )

  MsgInfo( "Arquivo INI criado e ocultado com sucesso!", "Sucesso" )

   ELSE

  MsgStop( "Erro ao criar o arquivo INI.", "Erro" )

   ENDIF

RETURN NIL
