STATIC FUNCTION EditarRegistro( oBrw, aData, oWnd )
   LOCAL nPos := oBrw:nArrayAt
   LOCAL aReg

   IF nPos == 0 .OR. Len( aData ) == 0
      MsgAlert( "Nenhum registro selecionado!", "Atenção" )
      RETURN NIL
   ENDIF

   aReg := AClone( aData[ nPos ] )
   

   IF FormularioCliente( aReg, .F., oWnd )
      aData[ nPos ] := aReg
      oBrw:RefreshCurrent()
      oBrw:MakeTotals()
   ENDIF
RETURN NIL

STATIC FUNCTION ExcluirRegistro( oBrw, aData )
   LOCAL nPos := oBrw:nArrayAt

   IF nPos == 0 .OR. Len( aData ) == 0
      MsgAlert( "Nenhum registro selecionado!", "Atenção" )
      RETURN NIL
   ENDIF

   IF MsgYesNo( "Deseja realmente excluir o cliente: " + AllTrim( aData[ nPos, 2 ] ) + "?", "Confirmação" )
      ADel( aData, nPos )
      ASize( aData, Len( aData ) - 1 )
      oBrw:Refresh()
      oBrw:MakeTotals()
   ENDIF

// ----------------------------------------------------------------------------
// VALIDADOR OFICIAL DE CPF (MODULO 11)
// ----------------------------------------------------------------------------
FUNCTION ValidarCPF( cCpf )
   LOCAL nSoma1 := 0, nSoma2 := 0, nDig1, nDig2, i
   cCpf := LimparNumero( cCpf )

   IF Len( cCpf ) != 11 .OR. cCpf == Replicate( SubStr( cCpf, 1, 1 ), 11 )
      RETURN .F.
   ENDIF

   FOR i := 1 TO 9
      nSoma1 += Val( SubStr( cCpf, i, 1 ) ) * ( 11 - i )
   NEXT
   nDig1 := ( nSoma1 * 10 ) % 11
   IF nDig1 == 10 .OR. nDig1 == 11
      nDig1 := 0
   ENDIF

   IF nDig1 != Val( SubStr( cCpf, 10, 1 ) )
      RETURN .F.
   ENDIF

   FOR i := 1 TO 10
      nSoma2 += Val( SubStr( cCpf, i, 1 ) ) * ( 12 - i )
   NEXT
   nDig2 := ( nSoma2 * 10 ) % 11
   IF nDig2 == 10 .OR. nDig2 == 11
      nDig2 := 0
   ENDIF

   IF nDig2 != Val( SubStr( cCpf, 11, 1 ) )
      RETURN .F.
   ENDIF

RETURN .T.

// ----------------------------------------------------------------------------
// VALIDADOR OFICIAL DE CNPJ (MODULO 11)
// ----------------------------------------------------------------------------
FUNCTION ValidarCNPJ( cCnpj )
   LOCAL aPesos1 := { 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2 }
   LOCAL aPesos2 := { 6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2 }
   LOCAL nSoma1 := 0, nSoma2 := 0, nResto1, nResto2, nDig1, nDig2, i
   cCnpj := LimparNumero( cCnpj )

   IF Len( cCnpj ) != 14 .OR. cCnpj == Replicate( SubStr( cCnpj, 1, 1 ), 14 )
      RETURN .F.
   ENDIF

   FOR i := 1 TO 12
      nSoma1 += Val( SubStr( cCnpj, i, 1 ) ) * aPesos1[i]
   NEXT
   nResto1 := nSoma1 % 11
   nDig1 := If( nResto1 < 2, 0, 11 - nResto1 )

   IF nDig1 != Val( SubStr( cCnpj, 13, 1 ) )
      RETURN .F.
   ENDIF

   FOR i := 1 TO 13
      nSoma2 += Val( SubStr( cCnpj, i, 1 ) ) * aPesos2[i]
   NEXT
   nResto2 := nSoma2 % 11
   nDig2 := If( nResto2 < 2, 0, 11 - nResto2 )

   IF nDig2 != Val( SubStr( cCnpj, 14, 1 ) )
      RETURN .F.
   ENDIF

RETURN .T.

      ENDIF
   NEXT

STATIC FUNCTION NextIDRaca()

   LOCAL nMax := 0
   LOCAL nRecno := recno()

   dbGoTop()
   DO WHILE !eof()
      IF Raca->ID > nMax
         nMax := Raca->ID
      ENDIF
      dbSkip( 1 )
   ENDDO

   dbGoTo( nRecno )

   RETURN nMax + 1
