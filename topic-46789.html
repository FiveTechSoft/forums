Dear Silvio,

This example uses a blurred snapshot of panel X, with a sharp panel Y above it. Closing Y removes the overlay and shows the original X again.

The important detail is to capture X BEFORE hiding it. Hide the live X while the overlay is displayed, otherwise its child controls can repaint over Y. The overlay is a sibling of X, not its child, so disabling X does not disable the Close button. The bitmap is drawn from bPainted so it survives repainting.

Code: Select all Collapse
#include "FiveWin.ch"

function Main()

   local oDlg, oX, oSay, oOpen
   local aLayer := { nil, 0, .T. }

   DEFINE DIALOG oDlg TITLE "TPanel: reversible snapshot blur" ;
      SIZE 620, 400 PIXEL TRUEPIXEL

   ACTIVATE DIALOG oDlg CENTERED ;
      ON INIT BuildPanel( oDlg, @oX, aLayer ) ;
      VALID ( CloseLayer( oX, aLayer ), .T. )

return nil

static function BuildPanel( oDlg, oX, aLayer )

   local oSay, oOpen

   oX := TPanel():New( 15, 15, 365, 595, oDlg )
   oX:SetColor( CLR_BLACK, RGB( 225, 235, 245 ) )

   @ 30, 25 SAY oSay PROMPT "This is panel X" ;
      SIZE 250, 25 PIXEL OF oX

   @ 90, 25 BUTTON oOpen PROMPT "Open panel Y" ;
      SIZE 150, 32 PIXEL OF oX ;
      ACTION ShowLayer( oDlg, oX, aLayer )

return nil

static function ShowLayer( oDlg, oX, aLayer )

   local hBmp, hBlur, oShade, oY, oSay, oClose

   if aLayer[ 1 ] != nil
      return nil
   endif

   hBmp := WndBitmap( oX:hWnd )
   if Empty( hBmp )
      MsgStop( "Could not capture panel X" )
      return nil
   endif

   hBlur := Blur( hBmp, 8 )
   DeleteObject( hBmp )
   if Empty( hBlur )
      MsgStop( "Blur needs a supported 24/32-bit bitmap" )
      return nil
   endif

   aLayer[ 2 ] := hBlur
   aLayer[ 3 ] := oX:IsEnabled()
   oX:Disable()
   // The live controls must not paint over the frozen snapshot.
   oX:Hide()

   // Sibling of X, not its child: disabling X must not disable Y.
   oShade := TPanel():New( oX:nTop, oX:nLeft, ;
      oX:nTop + oX:nHeight, oX:nLeft + oX:nWidth, oDlg )
   aLayer[ 1 ] := oShade
   oShade:bPainted := { | hDC | PalBmpDraw( hDC, 0, 0, aLayer[ 2 ] ) }

   // Y is sharp; only the captured background is blurred.
   oY := TPanel():New( 65, 140, 250, 440, oShade, , , .T. )
   oY:SetColor( CLR_BLACK, CLR_WHITE )
   @ 30, 25 SAY oSay PROMPT "Panel Y (not blurred)" ;
      SIZE 240, 25 PIXEL OF oY
   @ 105, 80 BUTTON oClose PROMPT "Close" ;
      SIZE 130, 32 PIXEL OF oY ;
      ACTION CloseLayer( oX, aLayer )

   oShade:GoTop()
   oShade:Refresh()
   oClose:SetFocus()

return nil

static function CloseLayer( oX, aLayer )

   if aLayer[ 1 ] != nil
      aLayer[ 1 ]:End()
      aLayer[ 1 ] := nil
   endif
   if ! Empty( aLayer[ 2 ] )
      DeleteObject( aLayer[ 2 ] )
      aLayer[ 2 ] := 0
   endif
   if oX != nil
      if aLayer[ 3 ]
         oX:Enable()
      endif
      oX:Show()
      oX:Refresh()
   endif

return .T.

This was compiled and visually tested with FWH 26.10 under Wine, not native Windows: ten open/close cycles, covering/uncovering the window, and closing the dialog with the overlay active. The example explicitly releases both bitmap handles and destroys the overlay.

This is a STATIC snapshot, not a live blur: changes to X while Y is open are not reflected in the blurred image. This fixed-size test starts with X visible and enabled; resizing and native Windows close/keyboard behavior have not been tested.
