// C:\FWH2603\SAMPLES\ERRBLOCK.PRG

#Include "FiveWin.Ch"

FUNCTION Main() 

   //In your Main()

   LOCAL oErr, bOldError := ErrorBlock( { | oErr | MyErrorHandler( oErr ) } )

   BEGIN SEQUENCE

  // Error, example

  ? 10 / 0

   RECOVER USING oErr

  ? "Error recovered:", oErr:description

   END SEQUENCE

   // Restaura o manipulador original
   ErrorBlock( bOldError )

RETURN NIL

STATIC FUNCTION MyErrorHandler( oErr )

? "A system error occurred:"

? "Description:", oErr:description

// Você pode registrar isso em um log e retornar o objeto de erro para um bloco RECOVER

BREAK( oErr )

RETURN NIL

// FIN / END
