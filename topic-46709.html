STATIC FUNCTION AplicaFiltro( nOpcStatus, cPorte, oBrw )

   LOCAL cFiltro := ""

   IF nOpcStatus == 2       // Livres
      cFiltro := 'STATUS == "L"'
   ELSEIF nOpcStatus == 3   // Ocupadas
      cFiltro := 'STATUS == "O"'
   ELSEIF nOpcStatus == 4   // Manutencao
      cFiltro := 'STATUS == "M"'
   ENDIF

   IF !Empty( cPorte ) .AND. AllTrim( cPorte ) != "TODOS"

  IF !Empty( cFiltro )
     cFiltro += ' .AND. '
  ENDIF

  cFiltro += 'AllTrim( Upper( TAMANHO ) ) == "' + ALLTRIM( UPPER( cPorte ) ) + '"'

   ENDIF

   IF Empty( cFiltro )

  DbClearFilter()

   ELSE

  SET FILTER TO &cFiltro

   ENDIF

   DBGOTOP()

   oBrw:Refresh()

RETURN NIL

STATIC FUNCTION BuscaRapida( cTexto, oBrw )

   LOCAL cTermo := Upper( AllTrim( cTexto ) )

   IF Empty( cTermo )

  DBGOTOP()

  oBrw:Refresh()

  RETURN NIL

   ENDIF

   // Se for codigo numerico, pesquisa no indice de codigo
   IF IsDigit( Left( cTermo, 1 ) )

  SET ORDER TO TAG NUMERO

  SEEK PadR( cTermo, 6 )

  IF !Found()

     SEEK cTermo

  ENDIF

   ELSE

  // Pesquisa por nome / descricao

  SET ORDER TO TAG DESCR

  SEEK cTermo

   ENDIF

   oBrw:Refresh()

RETURN NIL

// GRAVACAO NATIVA NO BANCO DBF
// ----------------------------------------------------------------------------
STATIC FUNCTION BaiaGravar( lInclusao, cNum, cDesc, cEsp, cTam, cStatus,     ;
                                       nVal, cPet, cTut, cObs, cObs2, cObs3, ;
                                       cObs4, cObs5, cObs6, oDlgEd, oBrw )

   IF Empty( cNum )

  MsgStop( "O codigo/numero da baia e obrigatorio!", "Aviso" )

  RETURN( .F. )

   ENDIF

   IF Empty( cDesc )

  MsgStop( "A descricao/identificacao da baia e obrigatoria!", "Aviso" )

  RETURN( .F. )

   ENDIF

   IF lInclusao

  // Validacao nativa com SEEK para evitar duplicidade de codigo
  SET ORDER TO TAG NUMERO
  DBSEEK( cNum )

  IF Found()

     MsgStop( "já existe uma baia cadastrada com o numero: " + STR( cNum ), "Atenção" )

     RETURN( .F. )

  ENDIF

  // Comando nativo de inclusao
  APPEND BLANK

  IF NetErr()

     MsgStop( "Falha de rede ao incluir o registro!", "Erro de Gravação" )

     RETURN( .F. )

  ENDIF

   ELSE

  // Bloqueio seguro para alteracao em rede
  IF !RLock()

     MsgStop( "O registro esta sendo editado por outro usuario na rede!", "Bloqueio" )

     RETURN( .F. )

  ENDIF

   ENDIF

   // Gravacao nativa campo a campo
   REPLACE FIELD->NUMERO    WITH cNum,            ;
           FIELD->DESCRICAO WITH cDesc,           ;
           FIELD->ESPECIE   WITH cEsp,            ;
           FIELD->TAMANHO   WITH cTam,            ;
           FIELD->STATUS    WITH cStatus,         ;
           FIELD->VALOR_DIA WITH nVal,            ;
           FIELD->PET_NOME  WITH cPet,            ;
           FIELD->TUTOR     WITH cTut,            ;
           FIELD->OBS       WITH cObs,            ;
           FIELD->OBS2      WITH cObs2,           ;
           FIELD->OBS3      WITH cObs3,           ;
           FIELD->OBS4      WITH cObs4,           ;
           FIELD->OBS5      WITH cObs5,           ;
           FIELD->OBS6      WITH cObs6

   DBUNLOCK()
   DBCOMMIT()

   GRAVACAO_EFETUADA_COM_SUCESSO()

   oBrw:Refresh()
   oDlgEd:End()

RETURN .T.

STATIC FUNCTION BaiaExcluir( oBrw )

     ENDIF

  ELSE

     MsgStop( "Registro travado na rede. Nao foi possivel excluir agora.", "Erro" )

  ENDIF

   ENDIF

STATIC FUNCTION BaiaImprimir( oBrw )

   LOCAL oPrn, oFontTit, oFontCab, oFontSub, oFontDet, oFontTot
   LOCAL nLinha := 1.0
   LOCAL nRecAtual := BAIAS->( RecNo() )
   LOCAL nTotBaias := 0, nTotLivres := 0, nTotOcup := 0, nTotManut := 0

   PRINT oPrn NAME "Relatório de Baias e Quartos - PetShop" PREVIEW

  DEFINE FONT oFontTit NAME "Arial" SIZE 0, -14 BOLD OF oPrn
  DEFINE FONT oFontSub NAME "Arial" SIZE 0, -10 OF oPrn
  DEFINE FONT oFontCab NAME "Courier New" SIZE 0, -10 BOLD OF oPrn
  DEFINE FONT oFontDet NAME "Courier New" SIZE 0, -9 OF oPrn
  DEFINE FONT oFontTot NAME "Courier New" SIZE 0, -10 BOLD OF oPrn

  PAGE
     // Cabecalho
     oPrn:Say( nLinha, 1.0, "PETSHOP HOTEL & HOSPEDAGEM - CONTROLE DE BAIAS/BOXES", oFontTit )
     nLinha += 0.5
     oPrn:Say( nLinha, 1.0, "Emissão: " + DToC( Date() ) + " as " + Time() + " | Filtro Atual: Ativo", oFontSub )
     nLinha += 0.5

     oPrn:Say( nLinha, 1.0, Replicate( "=", 110 ), oFontCab )
     nLinha += 0.4
     oPrn:Say( nLinha, 1.0, "NUM    IDENTIFICAÇÃO/NOME           ESPÉCIE   PORTE     STATUS     DIÁRIA R$  PET HOSPEDADO / TUTOR", oFontCab )
     nLinha += 0.35
     oPrn:Say( nLinha, 1.0, Replicate( "-", 110 ), oFontCab )
     nLinha += 0.45

     BAIAS->( DBGOTOP() )

     WHILE( .NOT. BAIAS->( EOF() ) )

        SYSREFRESH()

        IF !DELETED()

           nTotBaias++

           IF BAIAS->STATUS == "L"
              nTotLivres++
           ELSEIF BAIAS->STATUS == "O"
              nTotOcup++
           ELSE
              nTotManut++
           ENDIF

           oPrn:Say( nLinha, 1.0,  BAIAS->NUMERO, oFontDet )
           oPrn:Say( nLinha, 1.8,  Left( BAIAS->DESCRICAO, 28 ), oFontDet )
           oPrn:Say( nLinha, 4.6,  Left( BAIAS->ESPECIE, 8 ), oFontDet )
           oPrn:Say( nLinha, 5.6,  Left( BAIAS->TAMANHO, 9 ), oFontDet )
           oPrn:Say( nLinha, 6.7,  IF( BAIAS->STATUS == "L", "LIVRE", IF( BAIAS->STATUS == "O", "OCUPADO", "MANUT." ) ), oFontDet )
           oPrn:Say( nLinha, 7.8,  Transform( BAIAS->VALOR_DIA, "@E 9,999.99" ), oFontDet )
           oPrn:Say( nLinha, 9.0,  AllTrim( BAIAS->PET_NOME ) + IF( !EMPTY( BAIAS->TUTOR ), " (" + AllTrim( BAIAS->TUTOR ) + ")", "" ), oFontDet )

           nLinha += 0.40

           // Quebra de pagina se ultrapassar limite
           IF nLinha > 26.5

              oPrn:EndPage()

              oPrn:StartPage()

              nLinha := 1.5

           ENDIF

        ENDIF

        BAIAS->( DBSKIP() )

     ENDDO

     // Resumo Totalizador
     nLinha += 0.2

     oPrn:Say( nLinha, 1.0, Replicate( "=", 110 ), oFontCab )

     nLinha += 0.45

     oPrn:Say( nLinha, 1.0, "RESUMO GERAL: " + ;
                            "Total de Baias: " + AllTrim( Str( nTotBaias ) ) + "  |  " + ;
                            "Livres: " + AllTrim( Str( nTotLivres ) ) + "  |  " + ;
                            "Ocupadas: " + AllTrim( Str( nTotOcup ) ) + "  |  " + ;
                            "Em Manutencao: " + AllTrim( Str( nTotManut ) ), oFontTot )

  ENDPAGE

   ENDPRINT

   RELEASE FONT oFontTit, oFontSub, oFontCab, oFontDet, oFontTot

   // Restaura posicao original do cursor no DBF
   BAIAS->( DBGOTO( nRecAtual ) )

   oBrw:Refresh()

RETURN NIL
