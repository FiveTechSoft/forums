// MINI-MOTOR SecReport/SecSection : ejecuta la sintaxis DEFINE SECTION sobre
// TPrinter (unidades en cm, tamano de pagina real, apaisado automatico).
//=============================================================================
CLASS SecSection
   DATA oRpt, cHead, cHeadFont, nHeadSize, lHeadBold
   DATA bFoot, cFootFont, nFootSize, aCols
   METHOD New( oRpt ) CONSTRUCTOR
   METHOD SetHeader( c, f, s, b )
   METHOD SetFooter( b, f, s )
   METHOD AddColumn( c, w, b, a )
ENDCLASS

METHOD New( oRpt ) CLASS SecSection
   ::oRpt      := oRpt
   ::aCols     := {}
   ::cHead     := ""
   ::cHeadFont := "ARIAL"
   ::nHeadSize := 12
   ::lHeadBold := .F.
   ::bFoot     := NIL
   ::cFootFont := "ARIAL"
   ::nFootSize := 8
RETURN Self

METHOD SetHeader( c, f, s, b ) CLASS SecSection
   ::cHead := c
   IF ValType( f ) == "C" .AND. ! Empty( f ) ; ::cHeadFont := f ; ENDIF
   IF ValType( s ) == "N" .AND. s > 0        ; ::nHeadSize := s ; ENDIF
   ::lHeadBold := ( ValType( b ) == "L" .AND. b )
RETURN Self

METHOD SetFooter( b, f, s ) CLASS SecSection
   ::bFoot := b
   IF ValType( f ) == "C" .AND. ! Empty( f ) ; ::cFootFont := f ; ENDIF
   IF ValType( s ) == "N" .AND. s > 0        ; ::nFootSize := s ; ENDIF
RETURN Self

METHOD AddColumn( c, w, b, a ) CLASS SecSection
   DEFAULT a := DT_LEFT
   AAdd( ::aCols, { c, Max( 1, w ), b, a } )
RETURN Self

//----------------------------------------------------------------------------//
CLASS SecReport
   DATA cTitle, lPreview, aSecs, nPage
   DATA oPrn, oFontBody, oFontBold, oFontTitle, oFontHead, oFontFoot
   DATA lOpen, oLastSec, lLand, nPtBody
   DATA nLeft, nTop, nBottom, nFootRow, nPrintW, nRow, nBodyH
   DATA aColX, aColW
   METHOD New( cTitle, lPreview ) CONSTRUCTOR
   METHOD AddSection()
   METHOD PrintSection( oSec )
   METHOD End()
   METHOD StartSheet( oSec )
   METHOD HeadSheet( oSec )
   METHOD FootSheet( oSec )
ENDCLASS

METHOD New( cTitle, lPreview ) CLASS SecReport
   ::cTitle   := cTitle
   ::lPreview := ( ValType( lPreview ) == "L" .AND. lPreview )
   ::aSecs    := {}
   ::nPage    := 0
   ::lOpen    := .F.
   ::oLastSec := NIL
RETURN Self

METHOD AddSection() CLASS SecReport
   AAdd( ::aSecs, SecSection():New( Self ) )
RETURN ATail( ::aSecs )

METHOD PrintSection( oSec ) CLASS SecReport
   LOCAL i, aCol, cTxt, nPad
   IF ! ::lOpen
      ::StartSheet( oSec )
   ENDIF
   IF ::nRow + ::nBodyH > ::nBottom
      ::FootSheet( oSec )
      ::oPrn:EndPage()
      ::oPrn:StartPage()
      ::nPage++
      ::HeadSheet( oSec )
   ENDIF
   FOR i := 1 TO Len( oSec:aCols )
      aCol := oSec:aCols[ i ]
      cTxt := SecCStr( SecEval( aCol[ 3 ] ) )
      nPad := SecPad( aCol[ 4 ] )
      ::oPrn:CmSay( ::nRow, ::aColX[ i ], cTxt, ::oFontBody, ;
                    ::aColW[ i ],,, nPad )
   NEXT
   ::nRow += ::nBodyH
   ::oLastSec := oSec
RETURN Self

METHOD StartSheet( oSec ) CLASS SecReport
   LOCAL aGeo, nPxBody, nPxTitle, nPxHead, nPxFoot, cHF
   PRINTER ::oPrn NAME ::cTitle PREVIEW
   aGeo := SecColWidths( oSec:aCols, 1.2, 0 )  // solo nTot para decidir
   ::lLand := ( aGeo[ 3 ] > 100 )
   IF ::lLand
      ::oPrn:SetLandscape()
   ENDIF
   ::nLeft   := 1.2
   ::nTop    := 1.2
   ::nPrintW := ::oPrn:PageWidth( "CM" ) - 2.4
   ::nBottom := ::oPrn:PageHeight( "CM" ) - 2.2
   ::nFootRow := ::oPrn:PageHeight( "CM" ) - 1.4
   ::nPtBody := Min( 9, Max( 6, Int( ::nPrintW * 10 / aGeo[ 3 ] / 0.20 ) ) )
   ::nBodyH  := ::nPtBody * 0.067
   aGeo := SecColWidths( oSec:aCols, ::nLeft, ::nPrintW )
   ::aColX := aGeo[ 1 ]
   ::aColW := aGeo[ 2 ]
   nPxBody  := -::nPtBody
   nPxTitle := -( ::nPtBody + 6 )
   cHF      := oSec:cHeadFont
   nPxHead  := -Max( 8, oSec:nHeadSize )
   nPxFoot  := -Max( 6, oSec:nFootSize )
   DEFINE FONT ::oFontBody  NAME "ARIAL" SIZE 0, nPxBody  OF ::oPrn
   DEFINE FONT ::oFontBold  NAME "ARIAL" SIZE 0, nPxBody  BOLD OF ::oPrn
   DEFINE FONT ::oFontTitle NAME "ARIAL" SIZE 0, nPxTitle BOLD OF ::oPrn
   IF oSec:lHeadBold
      DEFINE FONT ::oFontHead NAME cHF SIZE 0, nPxHead BOLD OF ::oPrn
   ELSE
      DEFINE FONT ::oFontHead NAME cHF SIZE 0, nPxHead OF ::oPrn
   ENDIF
   DEFINE FONT ::oFontFoot NAME "ARIAL" SIZE 0, nPxFoot OF ::oPrn
   ::oPrn:StartPage()
   ::nPage  := 1
   ::lOpen  := .T.
   ::nRow   := ::nTop
   ::HeadSheet( oSec )
RETURN Self

METHOD HeadSheet( oSec ) CLASS SecReport
   LOCAL i, aCol
   ::oPrn:CmSay( ::nRow, ::nLeft, ::cTitle, ::oFontTitle, ::nPrintW,,, 2 )
   ::nRow += 0.85
   IF ! Empty( oSec:cHead )
      ::oPrn:CmSay( ::nRow, ::nLeft, oSec:cHead, ::oFontHead, ::nPrintW,,, 2 )
      ::nRow += 0.70
   ENDIF
   FOR i := 1 TO Len( oSec:aCols )
      aCol := oSec:aCols[ i ]
      ::oPrn:CmSay( ::nRow, ::aColX[ i ], aCol[ 1 ], ::oFontBold, ;
                    ::aColW[ i ],,, SecPad( aCol[ 4 ] ) )
   NEXT
   ::nRow += 0.60
RETURN Self

METHOD FootSheet( oSec ) CLASS SecReport
   LOCAL cTxt := ""
   IF ValType( oSec:bFoot ) == "B"
      cTxt := SecCStr( SecEval( oSec:bFoot ) )
   ENDIF
   IF ! Empty( cTxt )
      ::oPrn:CmSay( ::nFootRow, ::nLeft, cTxt, ::oFontFoot, ::nPrintW,,, 2 )
   ENDIF
RETURN Self

METHOD End() CLASS SecReport
   IF ::lOpen
      IF ::oLastSec != NIL
         ::FootSheet( ::oLastSec )
      ENDIF
      ::oPrn:EndPage()
      ::oPrn:Preview()
      ::oFontBody:End()
      ::oFontBold:End()
      ::oFontTitle:End()
      ::oFontHead:End()
      ::oFontFoot:End()
      ::lOpen := .F.
   ENDIF
RETURN NIL

//----------------------------------------------------------------------------//
// Funciones puras del mini-motor (testeables sin impresora via /testsec)
STATIC FUNCTION SecEval( b )
   LOCAL u := Eval( b )
   IF ValType( u ) == "B"
      u := Eval( u )
   ENDIF
RETURN u

STATIC FUNCTION SecCStr( u )
   LOCAL c
   DO CASE
   CASE ValType( u ) == "C" ; c := u
   CASE ValType( u ) == "D" ; c := DToC( u )
   CASE ValType( u ) == "N" ; c := LTrim( Str( u ) )
   CASE ValType( u ) == "L" ; c := If( u, "Sim", "Nao" )
   CASE ValType( u ) == "B" ; c := SecCStr( Eval( u ) )
   OTHERWISE                 ; c := ""
   ENDCASE
RETURN c

STATIC FUNCTION SecPad( nAlign )
   LOCAL aP := { 0, 2, 1 }  // DT_LEFT->PAD_LEFT, DT_CENTER->PAD_CENTER, DT_RIGHT->PAD_RIGHT
   DEFAULT nAlign := DT_LEFT
RETURN aP[ Min( Max( nAlign, 0 ), 2 ) + 1 ]

// aX/aW en cm + nTot ; nPrintW=0 -> solo calcula nTot (sin impresora)
STATIC FUNCTION SecColWidths( aCols, nLeft, nPrintW )
   LOCAL aX := {}, aW := {}, nTot := 0, i, nX, nW
   FOR i := 1 TO Len( aCols )
      nTot += aCols[ i ][ 2 ]
   NEXT
   nTot := Max( nTot, 1 )
   nX := nLeft
   FOR i := 1 TO Len( aCols )
      nW := nPrintW * aCols[ i ][ 2 ] / nTot
      AAdd( aX, nX )
      AAdd( aW, nW - 0.06 )
      nX += nW
   NEXT
RETURN { aX, aW, nTot }

STATIC FUNCTION ImprimeSintetico( nFiltro, nOrdem, lPreview )

   LOCAL oPrn, oFontEmp, oFontTit, oFontCab, oFontSub, oFontDet, oFontBld, oFontRod
   LOCAL oPenLinha, oPenGrossa

   LOCAL cFiltroTxt := "", cOrdemTxt := ""

   LOCAL lZebrado := .F.
   LOCAL cStatus := ""

   // Carrega e filtra os dados para uma matriz para permitir ordenacao customizada
   VET->( dbGoTop() )

   DO WHILE ! VET->( Eof() )

  cStatus := Upper( AllTrim( cValToChar( VET->ATIVO ) ) )

  IF Empty( cStatus )
     cStatus := "S" // Padrao ativo se nao definido
  ENDIF

  IF nFiltro == 2 .AND. cStatus != "S"
     VET->( dbSkip() )
     LOOP
  ENDIF
  IF nFiltro == 3 .AND. cStatus == "S"
     VET->( dbSkip() )
     LOOP
  ENDIF

  AAdd( aDados, { VET->CODIGO, ;                                       // 1
                  AllTrim( VET->NOME ), ;                              // 2
                  AllTrim( VET->CRMV ), ;                              // 3
                  AllTrim( VET->ESPECIALID ), ;                        // 4
                  AllTrim( VET->CPF ), ;                               // 5
                  If( !Empty( VET->WHATSAPP ), AllTrim( VET->WHATSAPP ), ;
                     If( !Empty( VET->CELULAR ), AllTrim( VET->CELULAR ), AllTrim( VET->FONE ) ) ), ; // 6
                  AllTrim( VET->CIDADE ) + If( !Empty( VET->UF ), "/" + AllTrim( VET->UF ), "" ), ; // 7
                  If( cStatus == "S", "Ativo", "Inativo" ) } )         // 8

  VET->( dbSkip() )
   ENDDO

   IF Len( aDados ) == 0
      MsgAlert( "Nenhum veterinario encontrado para o filtro selecionado!", "Aviso" )
      RETURN NIL
   ENDIF

   // Ordenacao da matriz
   DO CASE
      CASE nOrdem == 1 // Codigo
         ASort( aDados,,, { |x,y| x[ 1 ] < y[ 1 ] } )
         cOrdemTxt := "Codigo"
      CASE nOrdem == 2 // Nome

         cOrdemTxt := "Nome Alfab‚tico"
      CASE nOrdem == 3 // CRMV

         cOrdemTxt := "CRMV"

   ENDIF

   DEFINE FONT oFontBld    NAME "Arial" SIZE 0, -8.0 BOLD OF oPrn
   DEFINE FONT oFontRod    NAME "Arial" SIZE 0, -7.5     OF oPrn

   DEFINE PEN oPenGrossa WIDTH 1.8 COLOR CLR_TITULO OF oPrn

                            oPenLinha, oPenGrossa, @nRow, nPagina, cFiltroTxt, cOrdemTxt )
     ENDIF

     // Impressao das colunas da linha
     oPrn:CmSay( nRow, 1.0, StrZero( aDados[ i, 1 ], 6 ), oFontDet )
     oPrn:CmSay( nRow, 2.6, SubStr( aDados[ i, 2 ], 1, 40 ), oFontBld )

     nTotReg++
     IF aDados[ i, 8 ] == "Ativo"

  IF nRow > 17.0

CLASS SecSection
   DATA oRpt, cHead, cHeadFont, nHeadSize, lHeadBold
   DATA bFoot, cFootFont, nFootSize, aCols
   METHOD New( oRpt ) CONSTRUCTOR
   METHOD SetHeader( c, f, s, b )
   METHOD SetFooter( b, f, s )
   METHOD AddColumn( c, w, b, a )
ENDCLASS

METHOD New( oRpt ) CLASS SecSection
   ::oRpt      := oRpt
   ::aCols     := {}
   ::cHead     := ""
   ::cHeadFont := "ARIAL"
   ::nHeadSize := 12
   ::lHeadBold := .F.
   ::bFoot     := NIL
   ::cFootFont := "ARIAL"
   ::nFootSize := 8
RETURN Self

METHOD SetHeader( c, f, s, b ) CLASS SecSection
   ::cHead := c
   IF ValType( f ) == "C" .AND. ! Empty( f ) ; ::cHeadFont := f ; ENDIF
   IF ValType( s ) == "N" .AND. s > 0        ; ::nHeadSize := s ; ENDIF
   ::lHeadBold := ( ValType( b ) == "L" .AND. b )
RETURN Self

METHOD SetFooter( b, f, s ) CLASS SecSection
   ::bFoot := b
   IF ValType( f ) == "C" .AND. ! Empty( f ) ; ::cFootFont := f ; ENDIF
   IF ValType( s ) == "N" .AND. s > 0        ; ::nFootSize := s ; ENDIF
RETURN Self

METHOD AddColumn( c, w, b, a ) CLASS SecSection
   DEFAULT a := DT_LEFT
   AAdd( ::aCols, { c, Max( 1, w ), b, a } )
RETURN Self

      ::oPrn:StartPage()
      ::nPage++
      ::HeadSheet( oSec )

      aCol := oSec:aCols[ i ]

   NEXT

STATIC FUNCTION SecEval( b )

STATIC FUNCTION SecCStr( u )
   LOCAL c
   DO CASE
   CASE ValType( u ) == "C" ; c := u
   CASE ValType( u ) == "D" ; c := DToC( u )
   CASE ValType( u ) == "N" ; c := LTrim( Str( u ) )
   CASE ValType( u ) == "L" ; c := If( u, "Sim", "Nao" )
   CASE ValType( u ) == "B" ; c := SecCStr( Eval( u ) )
   OTHERWISE                 ; c := ""
   ENDCASE
RETURN c

STATIC FUNCTION SecPad( nAlign )
   LOCAL aP := { 0, 2, 1 }  // DT_LEFT->PAD_LEFT, DT_CENTER->PAD_CENTER, DT_RIGHT->PAD_RIGHT
   DEFAULT nAlign := DT_LEFT
