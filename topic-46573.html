   LOCAL oDlg, oFld

   DEFINE DIALOG oDlg TITLE "Date" ;
      FROM 0, 0 TO 360, 560 PIXEL

RETURN NIL


STATIC FUNCTION CreaCalendari( oFld, oCal1, oCal2, dArrivo, dPartenza )

   LOCAL oPage := oFld:aDialogs[ 1 ]
   LOCAL oCal, oSay1, oSay2


@ 250, 15 SAY oSay1 PROMPT "Arrivo: " + DToC( dArrivo ) ;
   OF oPage SIZE 220, 20 PIXEL

@ 250, 255 SAY oSay2 PROMPT "Partenza: " + DToC( dPartenza ) ;
   OF oPage SIZE 220, 20 PIXEL

@ 40, 15 CALENDAR oCal ;
   VAR dArrivo, dPartenza ;
   OF oPage ;
   SIZE 500, 140 PIXEL ;
   MULTISELECT ;
   ON CHANGE ( ;
      oSay1:SetText( "Arrivo: " + DToC( dArrivo ) ), ;
      oSay2:SetText( "Partenza: " + DToC( dPartenza ) ) )

oCal:SetMaxSelCount( 365 )  // selection

RETURN NIL
