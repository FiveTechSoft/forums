Brad,
Look at the TextOnImage() function in this working example:

Code: Select all Collapse
#Include "FiveWin.ch"

FUNCTION Main()
   LOCAL aGrad, oImg
   LOCAL Siz := { 600, 800 }, oFont1, oDlg
   SET DATE FORMAT "yyyy-mm-dd"
   aGrad := { { 0.30, CLR_WHITE, CLR_WHITE },{ 0.50, CLR_WHITE, CLR_WHITE } }
   DEFINE FONT oFont1 NAME "Verdana" SIZE 0,-20 BOLD
   DEFINE DIALOG oDlg FROM 0, 0 TO Siz[1], Siz[2] PIXEL TRUEPIXEL ;
      STYLE nOR( WS_POPUP, WS_BORDER ) GRADIENT aGrad
      @ 0,0 XIMAGE oImg FILE "D:\FWH\BITMAPS\PNGS\2.png" SIZE 0,0 OF oDlg NOBORDER
      oImg:bPainted:= { |hDC| TextOnImage(hdc,oFont1,"Text over img") }
      oImg:SetColor( CLR_GRAY, CLR_HRED ) // Added Mister Nages
   ACTIVATE DIALOG oDlg CENTERED
   RELEASE FONT oFont1
RETURN NIL
//-----------------------------------------------------------------//

Function  TextOnImage(hdc,oFont1,cCod)
           FW_SayText( hDC, cCod, {350,320,380,490 },,oFont1, CLR_GREEN,  , .f. )  // x1,y1 hasta x2,y2
           FW_SayText( hDC, DTOC(Date()), {380,340,410,470 },,oFont1, CLR_HBLUE,  , .f. )  // x1,y1 hasta x2,y2
return nil
//-----------------------------------------------------------------//

#include "FiveWin.ch"

FUNCTION TextOnImage()

   LOCAL hBmp, oFont, oLarg, oSmal
   LOCAL cSrcImage := "c:\fwh2603\bitmaps\olga1.jpg"
   LOCAL cDstImage := "Ambienty.jpg"
   LOCAL cText     := "Ambienty" + CRLF + "Paisagismo"

   IF FILE( "Ambienty.jpg" )
      DELETEFILE( "Ambienty.jpg" )
   ENDIF

   DEFINE FONT oLarg NAME "VERDANA" SIZE 0, - 50 BOLD
   DEFINE FONT oFont NAME "TAHOMA"  SIZE 0, - 30 BOLD
   DEFINE FONT oSmal NAME "TAHOMA"  SIZE 0, - 16 BOLD

   hBmp  :=  FW_MakeYourBitmap( 352, 450, < |hDC, w, h |

   FW_DrawImage( hDC, cSrcImage, { 0, 0, h, w } )
   FW_SayText( hDC, cText, { 0, 0, h - 30, w }, "B", { oLarg, oFont } )
   FW_SayText( hDC, "FWH Image Functions", { 30, 0, h, w }, "T", oSmal )
   RETURN nil
)

   FW_SaveImage( hBmp, cDstImage, 100 )

   XImage( cDstImage ) // Image modified

   // cleanup
   DeleteObject( hBmp )
   RELEASE FONT oLarg, oFont, oSmal

RETURN NIL

// FIN / END
