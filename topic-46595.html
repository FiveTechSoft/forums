Estimado Horacio,

Code: Select all Collapse
// file: factura_haru.prg  

#include "FiveWin.ch"  
  

REQUEST FWHARU  
  

function Main()  

   local oPrn  

   local oFontTitle, oFontBold, oFontText, oPen  
  

   TPrinter():lUseHaruPDF := .t.  
  

   PRINT oPrn NAME "Factura" PREVIEW FILE "factura.pdf"  
  

   DEFINE FONT oFontTitle NAME "Arial" SIZE 0, -18 BOLD OF oPrn  

   DEFINE FONT oFontBold  NAME "Arial" SIZE 0, -11 BOLD OF oPrn  

   DEFINE FONT oFontText  NAME "Arial" SIZE 0, -10 OF oPrn  

   DEFINE PEN  oPen WIDTH 11  
  

   PAGE  

      // Cabecera  

      oPrn:CmSay( 2.0, 2.0, "FACTURA", oFontTitle )  

      oPrn:CmSay( 2.7, 2.0, "Empresa de Ejemplo S.L.", oFontBold )  

      oPrn:CmSay( 3.1, 2.0, "CIF: B12345678", oFontText )  

      oPrn:CmSay( 3.5, 2.0, "Calle Principal 123 - Madrid", oFontText )  
  

  // Imagen / logo opcional  
  @ 1.2, 16.0 PRINT TO oPrn IMAGE "logo.png" SIZE 2.5, 2.0 INCHES  
  
  // Datos del cliente  
  oPrn:CmBox( 4.4, 2.0, 6.8, 20.0, oPen )  
  oPrn:CmSay( 4.6, 2.3, "Cliente:", oFontBold )  
  oPrn:CmSay( 5.1, 2.3, "Juan Perez", oFontText )  
  oPrn:CmSay( 5.6, 2.3, "Av. Siempre Viva 742", oFontText )  
  oPrn:CmSay( 6.1, 2.3, "28001 Madrid", oFontText )  
  
  // Datos de la factura  
  oPrn:CmBox( 7.2, 2.0, 8.9, 20.0, oPen )  
  oPrn:CmSay( 7.4, 2.3, "Factura Nro:", oFontBold )  
  oPrn:CmSay( 7.4, 6.0, "000123", oFontText )  
  oPrn:CmSay( 7.9, 2.3, "Fecha:", oFontBold )  
  oPrn:CmSay( 7.9, 6.0, "18/06/2026", oFontText )  
  oPrn:CmSay( 8.4, 2.3, "Vencimiento:", oFontBold )  
  oPrn:CmSay( 8.4, 6.0, "30/06/2026", oFontText )  
  
  // Observaciones  
  oPrn:CmBox( 9.2, 2.0, 10.4, 20.0, oPen )  
  oPrn:CmSay( 9.4, 2.3, "Observaciones:", oFontBold )  
  oPrn:CmSay( 9.9, 2.3, "Gracias por su compra.", oFontText )  
  
  // Tabla de líneas  
  oPrn:CmBox( 11.0, 2.0, 18.8, 20.0, oPen )  
  oPrn:CmLine( 11.6, 2.0, 11.6, 20.0, oPen )  
  oPrn:CmLine( 11.6, 12.0, 11.6, 20.0, oPen )  
  oPrn:CmLine( 11.6, 15.0, 11.6, 20.0, oPen )  
  oPrn:CmLine( 11.6, 17.0, 11.6, 20.0, oPen )  
  oPrn:CmLine( 11.6, 18.5, 11.6, 20.0, oPen )  
  
  oPrn:CmSay( 11.2, 2.3, "Cod.", oFontBold )  
  oPrn:CmSay( 11.2, 4.5, "Descripcion", oFontBold )  
  oPrn:CmSay( 11.2, 12.3, "Cant.", oFontBold )  
  oPrn:CmSay( 11.2, 15.3, "P.Unit", oFontBold )  
  oPrn:CmSay( 11.2, 17.3, "Imp.", oFontBold )  
  
  oPrn:CmSay( 12.0, 2.3, "A001", oFontText )  
  oPrn:CmSay( 12.0, 4.5, "Servicio de ejemplo", oFontText )  
  oPrn:CmSay( 12.0, 12.3, "2", oFontText )  
  oPrn:CmSay( 12.0, 15.3, "50.00", oFontText )  
  oPrn:CmSay( 12.0, 17.3, "100.00", oFontText )  
  
  oPrn:CmSay( 12.8, 2.3, "B002", oFontText )  
  oPrn:CmSay( 12.8, 4.5, "Producto adicional", oFontText )  
  oPrn:CmSay( 12.8, 12.3, "1", oFontText )  
  oPrn:CmSay( 12.8, 15.3, "30.00", oFontText )  
  oPrn:CmSay( 12.8, 17.3, "30.00", oFontText )  
  
  oPrn:CmSay( 13.6, 2.3, "C003", oFontText )  
  oPrn:CmSay( 13.6, 4.5, "Soporte", oFontText )  
  oPrn:CmSay( 13.6, 12.3, "3", oFontText )  
  oPrn:CmSay( 13.6, 15.3, "15.00", oFontText )  
  oPrn:CmSay( 13.6, 17.3, "45.00", oFontText )  
  
  // Totales  
  oPrn:CmBox( 19.2, 12.0, 23.2, 20.0, oPen )  
  oPrn:CmLine( 20.0, 12.0, 20.0, 20.0, oPen )  
  
  oPrn:CmSay( 19.5, 12.3, "Subtotal:", oFontBold )  
  oPrn:CmSay( 19.5, 17.1, "175.00", oFontText )  
  
  oPrn:CmSay( 20.4, 12.3, "IVA (21%):", oFontBold )  
  oPrn:CmSay( 20.4, 17.1, "36.75", oFontText )  
  
  oPrn:CmSay( 21.7, 12.3, "TOTAL:", oFontBold )  
  oPrn:CmSay( 21.7, 17.1, "211.75", oFontBold )  
   ENDPAGE  
  

   ENDPRINT  
  

return nil
