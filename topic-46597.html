METHOD GenerarIDR( nFactura ) CLASS VeriFactu

   local cIdr, cTs

   cTs := DToS( Date() ) + ;
          StrZero( Hour(), 2 ) + ;
          StrZero( Min(), 2 ) + ;
          StrZero( Sec(), 2 )

   cIdr := ::cNif + cTs + StrZero( nFactura, 4 )

   ::cUltimoIDR := cIdr

return cIdr

METHOD GenerarHuella( cIdr, cNumFact, cSerie, cFecha, cBase, cRecargo, cTotal ) CLASS VeriFactu

   local cData, cHash

   cData := cIdr + cNumFact + cSerie + cFecha + ;
            AllTrim( cBase ) + AllTrim( cRecargo ) + AllTrim( cTotal )

   // Simular SHA-256 (en produccion usar hb_sha256)
   cHash := ""
   local n, nSeed := 0
   for n := 1 to Len( cData )
      nSeed += Asc( SubStr( cData, n, 1 ) ) * n
   next

   // Generar hash de 64 caracteres hex
   cHash := ""
   local nVal
   for n := 1 to 64
      nVal := Mod( nSeed * n + n * 37, 16 )
      cHash += SubStr( "0123456789abcdef", nVal + 1, 1 )
      nSeed := Mod( nSeed + nVal, 256 )
   next

   ::cUltimaHuella := cHash

return cHash

METHOD CrearRegistro( aItems, aTotals, cNumFact, cFecha, cHora, ;
                      cClienteNif, cClienteNombre ) CLASS VeriFactu

   local hReg := {=>}, hCabecera := {=>}, hDatos := {=>}, hSistema := {=>}
   local hEmisor := {=>}, hDomicilio := {=>}
   local aLineas := {}, hLinea
   local cIdr, cHuella
   local nBase, nIva, nTotal, nRecargo
   local n, aItem

   // Validar datos minimos
   if Empty( ::cNif ) .or. Empty( ::cNombreRazon )
      ::Validar()
      return nil
   endif

   // Incrementar contador
   ::nContadorFactura++

   // Generar IDR
   cIdr := ::GenerarIDR( ::nContadorFactura )

   // Calcular totales de lineas
   nBase    := 0
   nIva     := 0
   nTotal   := 0
   nRecargo := 0

   // Crear lineas de facturacion
   for n := 1 to Len( aItems )
      aItem := aItems[ n ]
      hLinea := {=>}

  hLinea[ "NumeroLinea" ]           := n
  hLinea[ "DescripcionOperacion" ]  := aItem[ 2 ]  // descripcion
  hLinea[ "Cantidad" ]              := aItem[ 3 ]  // cantidad
  hLinea[ "PrecioUnitario" ]        := aItem[ 4 ]  // precio
  hLinea[ "ImporteTotal" ]          := Round( aItem[ 3 ] * aItem[ 4 ], 6 )
  hLinea[ "BaseImponibleOtributo" ] := Round( aItem[ 3 ] * aItem[ 4 ], 6 )

  // Impuesto
  hLinea[ "ClaveImpuesto" ]          := "01"  // IVA
  hLinea[ "ClaveRegimenEspecial" ]   := REG_GENERAL
  hLinea[ "TipoImpositivo" ]         := IVA_GENERAL
  hLinea[ "CuotaRepercutida" ]       := Round( aItem[ 3 ] * aItem[ 4 ] * IVA_GENERAL / 100, 6 )

  nBase += hLinea[ "BaseImponibleOtributo" ]
  nIva  += hLinea[ "CuotaRepercutida" ]

  AAdd( aLineas, hLinea )
   next

   nTotal := nBase + nIva + nRecargo

   // Cabecera
   hCabecera[ "IDREmisorFactura" ]      := cIdr

   hEmisor[ "NIFEmisor" ]              := ::cNif
   hEmisor[ "NombreRazonEmisor" ]       := ::cNombreRazon

   hDomicilio[ "DomicilioFiscal" ]      := ::cDomicilio
   hDomicilio[ "CodigoPostal" ]          := ::cCodigoPostal
   hDomicilio[ "Municipio" ]             := ::cMunicipio
   hDomicilio[ "Provincia" ]             := ::cProvincia
   hDomicilio[ "CodigoPais" ]            := ::cPais

   hEmisor[ "DomicilioEmisor" ]          := hDomicilio

   if ! Empty( ::cNIFRepresentante )
      hEmisor[ "NIFRepresentanteLegal" ] := ::cNIFRepresentante
   endif

   hCabecera[ "IDEmitente" ]            := hEmisor
   hCabecera[ "TipoFactura" ]           := If( Len( aItems ) <= 8, ;
                                             TIPO_SIMPLIF, TIPO_COMPLETA )
   hCabecera[ "TipoRectificativa" ]     := ""
   hCabecera[ "NumFactura" ]            := AllTrim( Str( ::nContadorFactura ) )
   hCabecera[ "SerieFactura" ]          := ::cSerieFactura
   hCabecera[ "FechaExpedicionFactura" ] := cFecha
   hCabecera[ "HoraExpedicionFactura" ] := cHora
   hCabecera[ "FacturadorSimplificado" ] := "S"

   // Datos de factura
   hDatos[ "DescripcionOperacion" ]     := ""
   hDatos[ "BaseImponible" ]            := Round( nBase, 6 )
   hDatos[ "CuotaRecargo" ]             := Round( nRecargo, 6 )
   hDatos[ "TotalFactura" ]             := Round( nTotal, 6 )
   hDatos[ "ImporteTotalPymes" ]        := Round( nTotal, 6 )

   hDatos[ "DatosRetencion" ]           := {=>}
   hDatos[ "DatosRetencion" ][ "BaseRetencion" ] := 0
   hDatos[ "DatosRetencion" ][ "PorcentajeRetencion" ] := 0
   hDatos[ "DatosRetencion" ][ "ImporteRetencion" ] := 0

   hDatos[ "LineasFactura" ]            := aLineas

   // Sistema VeriFactu
   cHuella := ::GenerarHuella( cIdr, ;
              AllTrim( Str( ::nContadorFactura ) ), ;
              ::cSerieFactura, cFecha, ;
              AllTrim( Trans( nBase, "@E 999999.999999" )), ;
              AllTrim( Trans( nRecargo, "@E 999999.999999" )), ;
              AllTrim( Trans( nTotal, "@E 999999.999999" )) )

   hSistema[ "Truncamiento" ]           := "N"
   hSistema[ "Encadenamiento" ]         := {=>}
   hSistema[ "Encadenamiento" ][ "PrimerRegistro" ] := ;
      If( Len( ::aRegistros ) == 0, "S", "N" )

   if Len( ::aRegistros ) > 0
      hSistema[ "Encadenamiento" ][ "IDRRegistroAnterior" ] := ;
         ::aRegistros[ Len( ::aRegistros ), "Cabecera", "IDREmisorFactura" ]
      hSistema[ "Encadenamiento" ][ "HuellaRegistroAnterior" ] := ;
         ::cUltimaHuella
   endif

   hSistema[ "SistemaUtilizado" ]       := "01"
   hSistema[ "Huella" ]                 := cHuella
   hSistema[ "VersionSCHEMA" ]          := "1.0"
   hSistema[ "VersionSOFTWARE" ]        := ::cSoftwareVersion
   hSistema[ "NombreSOFTWARE" ]         := ::cSoftwareName
   hSistema[ "NumeroSerialSoftware" ]   := StrZero( ::nContadorFactura, 10 )

   // Ensamblar registro
   hReg[ "Cabecera" ]      := hCabecera
   hReg[ "DatosFactura" ]  := hDatos
   hReg[ "SistemaVeriFactu" ] := hSistema

   // Metadata adicional
   hReg[ "_clienteNif" ]     := cClienteNif
   hReg[ "_clienteNombre" ]  := cClienteNombre
   hReg[ "_timestamp" ]      := DToC( Date() ) + " " + Time()
   hReg[ "_modo" ]           := If( ::lModoPrueba, "PRUEBA", "PRODUCCION" )

   // Guardar en registro
   AAdd( ::aRegistros, hReg )

return hReg

METHOD EnviarAEAT( hReg ) CLASS VeriFactu

   local cXml, cResponse := ""
   local oHttp, oErr

   if ::lModoPrueba
      // En modo prueba, simular respuesta exitosa
      ::LogRegistro( hReg, "ENVIADO_PRUEBA" )
      return { "ok" => .T., "estado" => "PRUEBA", "mensaje" => "Registro generado (modo prueba)" }
   endif

   // Generar XML
   cXml := ::RegistroToXml( hReg )

   // Enviar a AEAT via HTTP POST
   BEGIN SEQUENCE
      oHttp := TIpClientHttp():New( ::cEndpoint )
      oHttp:setPostFields( cXml )
      oHttp:addHeader( "Content-Type", "application/xml" )
      oHttp:addHeader( "Accept", "application/xml" )

  // En produccion: agregar certificado digital
  // oHttp:setCertFile( ::cCertificadoRuta, ::cCertificadoPass )

  cResponse := oHttp:post()

  ::LogRegistro( hReg, "ENVIADO" )

  return { "ok" => .T., "estado" => "ENVIADO", "respuesta" => cResponse }

   RECOVER USING oErr
      ::LogRegistro( hReg, "ERROR: " + oErr:Description )
      return { "ok" => .F., "estado" => "ERROR", "mensaje" => oErr:Description }
   END

return { "ok" => .F., "estado" => "ERROR", "mensaje" => "Desconocido" }

METHOD Validar() CLASS VeriFactu

   local aErrores := {}

   if Empty( ::cNif )
      AAdd( aErrores, "NIF del emisor no configurado" )
   endif

   if Empty( ::cNombreRazon )
      AAdd( aErrores, "Nombre/Razon social no configurado" )
   endif

   if Empty( ::cDomicilio )
      AAdd( aErrores, "Domicilio no configurado" )
   endif

   if Empty( ::cCodigoPostal )
      AAdd( aErrores, "Codigo postal no configurado" )
   endif

   if Empty( ::cMunicipio )
      AAdd( aErrores, "Municipio no configurado" )
   endif

   if Empty( ::cProvincia )
      AAdd( aErrores, "Provincia no configurado" )
   endif

return aErrores

METHOD GetEstado() CLASS VeriFactu

   local hEstado := {=>}

   hEstado[ "modo" ]             := If( ::lModoPrueba, "PRUEBA", "PRODUCCION" )
   hEstado[ "nif" ]              := ::cNif
   hEstado[ "nombre" ]           := ::cNombreRazon
   hEstado[ "software" ]         := ::cSoftwareName + " v" + ::cSoftwareVersion
   hEstado[ "registros" ]        := Len( ::aRegistros )
   hEstado[ "ultimaHuella" ]     := ::cUltimaHuella
   hEstado[ "ultimoIDR" ]        := ::cUltimoIDR
   hEstado[ "serie" ]            := ::cSerieFactura
   hEstado[ "siguienteFactura" ] := ::nContadorFactura + 1
   hEstado[ "endpoint" ]         := ::cEndpoint
   hEstado[ "certificado" ]      := If( ! Empty( ::cCertificadoRuta ), "Configurado", "No configurado" )

return hEstado

METHOD LogRegistro( hReg, cEstado ) CLASS VeriFactu

   local cLinea

   cLinea := DToC( Date() ) + " " + Time() + " | " + ;
             AllTrim( hReg[ "Cabecera", "IDREmisorFactura" ] ) + " | " + ;
             "F" + AllTrim( hReg[ "Cabecera", "NumFactura" ] ) + ;
             "/" + AllTrim( hReg[ "Cabecera", "SerieFactura" ] ) + " | " + ;
             cEstado + " | " + ;
             "$" + AllTrim( Trans( hReg[ "DatosFactura", "TotalFactura" ], "@E 999,999.99" ) )

   LogFile( "verifactu.log", { cLinea } )

return nil

function VF_Facturar( oVF, aItems, cNumFact, cClienteNif, cClienteNombre )

   local hReg, cXml, hResult
   local nSub := 0, nIva, nTotal
   local n

   // Calcular totales
   for n := 1 to Len( aItems )
      nSub += aItems[ n, 3 ] * aItems[ n, 4 ]
   next
   nIva   := Round( nSub * IVA_GENERAL / 100, 2 )
   nTotal := nSub + nIva

   // Crear registro VeriFactu
   hReg := oVF:CrearRegistro( aItems, ;
           { "subtotal" => nSub, "iva" => nIva, "total" => nTotal }, ;
           cNumFact, DToC( Date() ), Time(), ;
           cClienteNif, cClienteNombre )

   if hReg == nil
      return { "ok" => .F., "error" => "Error creando registro" }
   endif

   // Generar XML
   cXml := oVF:RegistroToXml( hReg )

   // Enviar a AEAT
   hResult := oVF:EnviarAEAT( hReg )

   hResult[ "xml" ] := cXml
   hResult[ "registro" ] := hReg

return hResult

function VF_GuardarXml( cXml, cRuta )

   local lOk := .F.

   BEGIN SEQUENCE
      MemoWrit( cRuta, cXml )
      lOk := .T.
   RECOVER
      lOk := .F.
   END

return lOk
