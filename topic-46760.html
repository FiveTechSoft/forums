Hello,

I am experimenting with an old idea: forms where every field is an input, a formula or a decision tree, all connected and recalculated in dependency order, a bit in the spirit of the form and decision-tree tools of the early 90s. This is an original prototype written in Harbour and FWH. It uses no third-party assets and it is not file-compatible with any product.

📎 objectvision_wine.png

The screenshot is a real run of the code below, built with Harbour + FWH and executed under Wine 11.19 on Linux (not tested on real Windows, not tested with xHarbour).

What you see: an MDI frame with six real MDICHILD windows (Actions, Form, Tree, Properties, Status, Components). The form has real GETs. The button "Sumar" is linked to the field "Cantidad": when it is pressed, its action tree runs first, then the link adds 1 to Cantidad, and the subtotal, net and total are recalculated.

What works in my tests:

  • Creating input, formula, decision-tree and button fields from the Components window.
  • Editing branches, operator, threshold and result of a tree (first matching branch wins).
  • Dragging a field onto the root of a tree to change the field it decides on (cycles are rejected, F12 undoes).
  • Dragging a button onto an input field to link its "on click" event to that field (set to zero or add 1). The link is drawn as an arrow, saved to a small INI file, and F12 undoes it.
  • Moving, resizing and minimizing the windows, with the layout saved.
  • Saving orders to a DBF.

Limits: it is a prototype. One event ("al pulsar"), one link per button, two link actions. The formula engine only evaluates numbers, field names, + - * / and parentheses (text is never evaluated as code). After resizing a window the content is clipped. UI texts are in Spanish.

/* dtree.prg - field-network engine for "FWH Decision Forms" (original code, pure Harbour, no FWH, no macros).
   Fields: inputs (set by the user), formula fields (own mini evaluator: + - * / ( ) numbers and field names)
   and tree fields (first matching branch wins). Targets are recalculated in dependency order; cycles are rejected.
   Text is never evaluated as Harbour code (no &, no Eval of strings).                                          */

#include "hbclass.ch"

FUNCTION DT_Cond( cOp, xArg, xVal )
   DO CASE
   CASE cOp == "otherwise" ; RETURN .T.
   CASE cOp == ">"         ; RETURN xVal >  xArg
   CASE cOp == ">="        ; RETURN xVal >= xArg
   CASE cOp == "<"         ; RETURN xVal <  xArg
   CASE cOp == "<="        ; RETURN xVal <= xArg
   CASE cOp == "="         ; RETURN xVal == xArg
   CASE cOp == "<>"        ; RETURN !( xVal == xArg )
   CASE cOp == "between"   ; RETURN xVal >= xArg[ 1 ] .AND. xVal <= xArg[ 2 ]
   ENDCASE
RETURN .F.

/* ---- mini expression evaluator --------------------------------------------------------------------- */
STATIC FUNCTION EV_Tok( cE )
   LOCAL aT := {}, n := 1, nL := Len( cE ), c, nS
   DO WHILE n <= nL
      c := SubStr( cE, n, 1 )
      DO CASE
      CASE c == " "
         n++
      CASE c $ "+-*/()"
         AAdd( aT, { "op", c } ); n++
      CASE c $ "0123456789."
         nS := n
         DO WHILE n <= nL .AND. SubStr( cE, n, 1 ) $ "0123456789."; n++; ENDDO
         AAdd( aT, { "n", Val( SubStr( cE, nS, n - nS ) ), SubStr( cE, nS, n - nS ) } )
      CASE ( c >= "A" .AND. c <= "Z" ) .OR. ( c >= "a" .AND. c <= "z" ) .OR. c == "_"
         nS := n
         DO WHILE n <= nL .AND. ( SubStr( cE, n, 1 ) $ "0123456789_" .OR. ( Upper( SubStr( cE, n, 1 ) ) >= "A" .AND. Upper( SubStr( cE, n, 1 ) ) <= "Z" ) ); n++; ENDDO
         AAdd( aT, { "id", SubStr( cE, nS, n - nS ) } )
      OTHERWISE
         RETURN NIL
      ENDCASE
   ENDDO
RETURN aT

/* aS := { aTokens, nPos, hFields, lErr } */
STATIC FUNCTION EV_Peek( aS )
RETURN iif( aS[ 2 ] <= Len( aS[ 1 ] ), aS[ 1 ][ aS[ 2 ] ], NIL )

STATIC FUNCTION EV_Prim( aS )
   LOCAL t := EV_Peek( aS ), x := 0
   IF t == NIL; aS[ 4 ] := .T.; RETURN 0; ENDIF
   aS[ 2 ]++
   DO CASE
   CASE t[ 1 ] == "n"
      RETURN t[ 2 ]
   CASE t[ 1 ] == "id"
      IF t[ 2 ] $ aS[ 3 ] .AND. ValType( aS[ 3 ][ t[ 2 ] ] ) == "N"
         RETURN aS[ 3 ][ t[ 2 ] ]
      ENDIF
      aS[ 4 ] := .T.; RETURN 0
   CASE t[ 1 ] == "op" .AND. t[ 2 ] == "("
      x := EV_Expr( aS )
      t := EV_Peek( aS )
      IF t == NIL .OR. ! ( t[ 1 ] == "op" .AND. t[ 2 ] == ")" ); aS[ 4 ] := .T.; RETURN 0; ENDIF
      aS[ 2 ]++
      RETURN x
   ENDCASE
   aS[ 4 ] := .T.
RETURN 0

STATIC FUNCTION EV_Unary( aS )
   LOCAL t := EV_Peek( aS )
   IF t != NIL .AND. t[ 1 ] == "op" .AND. t[ 2 ] == "-"
      aS[ 2 ]++
      RETURN - EV_Unary( aS )
   ENDIF
RETURN EV_Prim( aS )

STATIC FUNCTION EV_Term( aS )
   LOCAL x := EV_Unary( aS ), t, y
   DO WHILE ! aS[ 4 ]
      t := EV_Peek( aS )
      IF t == NIL .OR. t[ 1 ] != "op" .OR. ! ( t[ 2 ] $ "*/" ); EXIT; ENDIF
      aS[ 2 ]++
      y := EV_Unary( aS )
      IF t[ 2 ] == "*"
         x := x * y
      ELSEIF y == 0
         aS[ 4 ] := .T.
      ELSE
         x := x / y
      ENDIF
   ENDDO
RETURN x

STATIC FUNCTION EV_Expr( aS )
   LOCAL x := EV_Term( aS ), t, y
   DO WHILE ! aS[ 4 ]
      t := EV_Peek( aS )
      IF t == NIL .OR. t[ 1 ] != "op" .OR. ! ( t[ 2 ] $ "+-" ); EXIT; ENDIF
      aS[ 2 ]++
      y := EV_Term( aS )
      x := iif( t[ 2 ] == "+", x + y, x - y )
   ENDDO
RETURN x

/* DT_Eval( cExpr, hFields, @lErr ) -> number (0 when lErr: bad syntax, unknown field, division by zero) */
FUNCTION DT_Eval( cE, hF, lErr )
   LOCAL aT := EV_Tok( cE ), aS, x
   IF aT == NIL .OR. Len( aT ) == 0; lErr := .T.; RETURN 0; ENDIF
   aS := { aT, 1, hF, .F. }
   x := EV_Expr( aS )
   IF aS[ 2 ] <= Len( aT ); aS[ 4 ] := .T.; ENDIF
   lErr := aS[ 4 ]
RETURN iif( lErr, 0, x )

/* Canonical text: single spaces between tokens, field names spelled as in aKnown (call after DT_Check). */
FUNCTION DT_Canon( cE, aKnown )
   LOCAL aT := EV_Tok( cE ), c := "", t, k
   IF aT == NIL; RETURN cE; ENDIF
   FOR EACH t IN aT
      IF t[ 1 ] == "id"
         k := AScan( aKnown, { | x | Upper( x ) == Upper( t[ 2 ] ) } )
         c += iif( Empty( c ), "", " " ) + iif( k > 0, aKnown[ k ], t[ 2 ] )
      ELSEIF t[ 1 ] == "n"
         c += iif( Empty( c ), "", " " ) + t[ 3 ]
      ELSE
         c += iif( Empty( c ), "", " " ) + t[ 2 ]
      ENDIF
   NEXT
RETURN c

/* names used by an expression (NIL when the text has invalid characters) */
FUNCTION DT_Names( cE )
   LOCAL aT := EV_Tok( cE ), aN := {}, t
   IF aT == NIL; RETURN NIL; ENDIF
   FOR EACH t IN aT
      IF t[ 1 ] == "id"; AAdd( aN, t[ 2 ] ); ENDIF
   NEXT
RETURN aN

/* Static check of an expression against known field names: returns "" or the reason.
   Division by zero is a run-time condition (values), not checked here, except literal 1/0 style constants. */
FUNCTION DT_Check( cE, aKnown )
   LOCAL aN, x, hT := { => }, lE := .F.
   IF Empty( AllTrim( cE ) ); RETURN "la fórmula está vacía"; ENDIF
   aN := DT_Names( cE )
   IF aN == NIL; RETURN "carácter no permitido (solo números, nombres de campo, + - * / y paréntesis)"; ENDIF
   FOR EACH x IN aN
      IF AScan( aKnown, { | k | Upper( k ) == Upper( x ) } ) == 0; RETURN "campo desconocido: " + x; ENDIF
   NEXT
   FOR EACH x IN aKnown; hT[ x ] := 7.3; NEXT
   DT_Eval( DT_Canon( cE, aKnown ), hT, @lE )
   IF lE; RETURN "sintaxis no válida o división por cero"; ENDIF
RETURN ""

/* Returns "" if ok, otherwise the reason (the node is not added) */
METHOD Rule( cTarget, cTest, aBranches ) CLASS TDForm
   LOCAL cErr
   AAdd( ::aRules, { cTarget, cTest, aBranches } )
   cErr := ::Order()
   IF ! Empty( cErr ); ASize( ::aRules, Len( ::aRules ) - 1 ); ::Order(); ENDIF
RETURN cErr

METHOD Formula( cTarget, cExpr ) CLASS TDForm
   LOCAL cErr, aN := DT_Names( cExpr )
   IF aN == NIL .OR. Len( aN ) == 0 .AND. Empty( cExpr ); RETURN "formula invalida para " + cTarget; ENDIF
   AAdd( ::aForms, { cTarget, cExpr } )
   cErr := ::Order()
   IF ! Empty( cErr ); ASize( ::aForms, Len( ::aForms ) - 1 ); ::Order(); ENDIF
RETURN cErr

/* Dependency order (Kahn). Returns "" or "ciclo: A, B". Also rejects two nodes with the same target. */
METHOD Order() CLASS TDForm
   LOCAL aNode := {}, aDone := {}, aT := {}, n, m, k, lProg, aDep, cErr := "", aOut := {}, x, lOk, c
   FOR n := 1 TO Len( ::aRules ); AAdd( aNode, { "R", n, ::aRules[ n ][ 1 ], { ::aRules[ n ][ 2 ] } } ); NEXT
   FOR n := 1 TO Len( ::aForms ); AAdd( aNode, { "F", n, ::aForms[ n ][ 1 ], DT_Names( ::aForms[ n ][ 2 ] ) } ); NEXT
   FOR n := 1 TO Len( aNode )
      IF AScan( aT, aNode[ n ][ 3 ] ) > 0; RETURN "el campo " + aNode[ n ][ 3 ] + " tiene dos definiciones"; ENDIF
      AAdd( aT, aNode[ n ][ 3 ] )
   NEXT
   aDone := Array( Len( aNode ) ); AFill( aDone, .F. )
   DO WHILE Len( aOut ) < Len( aNode )
      lProg := .F.
      FOR n := 1 TO Len( aNode )
         IF aDone[ n ]; LOOP; ENDIF
         lOk := .T.
         FOR EACH x IN aNode[ n ][ 4 ]
            k := AScan( aT, x )
            IF k > 0 .AND. ! aDone[ k ]; lOk := .F.; EXIT; ENDIF
         NEXT
         IF lOk; aDone[ n ] := .T.; AAdd( aOut, { aNode[ n ][ 1 ], aNode[ n ][ 2 ] } ); lProg := .T.; ENDIF
      NEXT
      IF ! lProg
         FOR n := 1 TO Len( aNode )
            IF ! aDone[ n ]; cErr += iif( Empty( cErr ), "", ", " ) + aNode[ n ][ 3 ]; ENDIF
         NEXT
         RETURN "ciclo entre campos: " + cErr
      ENDIF
   ENDDO
   ::aOrder := aOut
RETURN ""

/* Replace the expression of EXISTING formula fields. aPairs = { { cTarget, cExpr }, ... }. Atomic: returns "" or the reason and leaves the model unchanged.
   Field names are matched ignoring case and stored with their canonical spelling. */
METHOD SetFormulas( aPairs ) CLASS TDForm
   LOCAL aOld := {}, p, k, cErr := "", aKnown := hb_HKeys( ::hF ), n
   FOR EACH p IN aPairs
      k := AScan( ::aForms, { | f | Upper( f[ 1 ] ) == Upper( p[ 1 ] ) } )
      IF k == 0; RETURN "no existe el campo calculado " + p[ 1 ]; ENDIF
      cErr := DT_Check( p[ 2 ], aKnown )
      IF ! Empty( cErr ); RETURN p[ 1 ] + ": " + cErr; ENDIF
   NEXT
   FOR n := 1 TO Len( ::aForms ); AAdd( aOld, ::aForms[ n ][ 2 ] ); NEXT
   FOR EACH p IN aPairs
      k := AScan( ::aForms, { | f | Upper( f[ 1 ] ) == Upper( p[ 1 ] ) } )
      ::aForms[ k ][ 2 ] := DT_Canon( p[ 2 ], aKnown )
   NEXT
   cErr := ::Order()
   IF ! Empty( cErr )
      FOR n := 1 TO Len( ::aForms ); ::aForms[ n ][ 2 ] := aOld[ n ]; NEXT
      ::Order()
      RETURN cErr
   ENDIF
   ::Recalc()
RETURN ""

METHOD Get( cName ) CLASS TDForm

/* One pass in dependency order is enough (no cycles). Formula results are rounded to 2 decimals. */
METHOD Recalc() CLASS TDForm
   LOCAL a, aR, aB, lE, xV
   ::lErr := .F.
   FOR EACH a IN ::aOrder
      IF a[ 1 ] == "R"
         aR := ::aRules[ a[ 2 ] ]
         IF aR[ 2 ] $ ::hF
            FOR EACH aB IN aR[ 3 ]
               IF DT_Cond( aB[ 1 ], aB[ 2 ], ::hF[ aR[ 2 ] ] )
                  ::hF[ aR[ 1 ] ] := aB[ 3 ][ 2 ]
                  EXIT
               ENDIF
            NEXT
         ENDIF
      ELSE
         lE := .F.
         xV := DT_Eval( ::aForms[ a[ 2 ] ][ 2 ], ::hF, @lE )
         IF lE; ::lErr := .T.; ENDIF
         ::hF[ ::aForms[ a[ 2 ] ][ 1 ] ] := Round( xV, 2 )
      ENDIF
   NEXT
RETURN Self

/* branches: { cOp, xThr, { "const", nRes } }. Returns "" when coherent (>= and > descending together, or <= and < ascending together; "otherwise" last). */
FUNCTION DT_ValidateBranches( aB )
   LOCAL n, nDesc := 0, nAsc := 0, nPrev := NIL, lDesc
   FOR n := 1 TO Len( aB )
      IF aB[ n ][ 1 ] == "otherwise"
         IF n != Len( aB ); RETURN "'en otro caso' debe ser la última rama"; ENDIF
         LOOP
      ENDIF
      IF aB[ n ][ 1 ] $ ">=|>"; nDesc++; ELSE; nAsc++; ENDIF
   NEXT
   IF Len( aB ) < 2 .OR. aB[ Len( aB ) ][ 1 ] != "otherwise"; RETURN "Falta la rama final 'en otro caso'"; ENDIF
   IF nDesc > 0 .AND. nAsc > 0
      RETURN "Los operadores deben ir en el mismo sentido: >= y > juntos, o <= y < juntos"
   ENDIF
   lDesc := ( nDesc > 0 )
   FOR n := 1 TO Len( aB ) - 1
      IF nPrev != NIL
         IF lDesc .AND. aB[ n ][ 2 ] >= nPrev
            RETURN "El umbral de la rama " + LTrim( Str( n ) ) + " debe ser menor que el de la rama " + LTrim( Str( n - 1 ) ) + " (se evalúa en orden)"
         ENDIF
         IF ! lDesc .AND. aB[ n ][ 2 ] <= nPrev
            RETURN "El umbral de la rama " + LTrim( Str( n ) ) + " debe ser mayor que el de la rama " + LTrim( Str( n - 1 ) ) + " (se evalúa en orden)"
         ENDIF
      ENDIF
      nPrev := aB[ n ][ 2 ]
   NEXT
RETURN ""
