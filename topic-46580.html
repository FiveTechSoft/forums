Cambiar color &quot;thinking&quot; en Claude Code

Posted: Fri Jun 12, 2026 05:37 AM

Desde Claude Code:

"Crea un pixel shader para Windows Terminal que detecte texto de luminosidad media y lo ilumine"

Y el ajuste fino: "baja/subela intensidad del shader al X%".

theme dark + foreground verde #16C60C + shader 20%

boost-dim.hlsl

Code: Select all Collapse
Texture2D shaderTexture;
SamplerState samplerState;

cbuffer PixelShaderSettings {
    float  Time;
    float  Scale;
    float2 Resolution;
    float4 Background;
};

float4 main(float4 pos : SV_POSITION, float2 tex : TEXCOORD0) : SV_TARGET
{
    float4 color = shaderTexture.Sample(samplerState, tex);
    float luminance = dot(color.rgb, float3(0.299, 0.587, 0.114));

// Dim text falls in mid-luminance range on dark background
float dimZone = smoothstep(0.15, 0.35, luminance) * smoothstep(0.75, 0.55, luminance);

// Boost dim text: 0.0=off, 1.0=max
float boosted = lerp(luminance, 1.0, dimZone * 0.8);
color.rgb *= boosted / max(luminance, 0.001);

return color;
}
