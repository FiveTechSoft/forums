cGetFile32() Tambien NO FUNCIONA.

Code: Select all Collapse
// C:\FWH\SAMPLES\GETFIL32.PRG

#include "FiveWin.ch"
#Include "Directry.ch"

FUNCTION Main()

   LOCAL cDirExe := GETCURDIR()
   LOCAL cNewDir := cDirExe // "C:\MyFolder"
   LOCAL nResult, cFile
   LOCAL cFiltro, cArquivo

   IF SUBS( cDirExe, LEN( ALLTRIM(cDirExe ) ) , 1 ) = "\"
      cDirExe := SUBS( cDirExe, 1 , LEN( ALLTRIM(cDirExe ) ) - 1 )
   ENDIF

   nResult := DirChange( cNewDir ) // LCHDIR( cDirExe )

   IF nResult == 0

  MsgInfo( "Directory changed successfully to: " + cNewDir )

   ELSE

  MsgStop( "Error: Could not change directory. Check if the path exists." )

   ENDIF

   cFiltro := "Arquivos JPG (*.jpg) | *.jpg |" + ;
              "Todos os Arquivos (*.*) | *.* |"

   cArquivo := cGetFile32( cFiltro, "Selecione o arquivo JPG para importar", 1, "C:\PETSHOP" )

   IF Empty( cArquivo )
      MsgStop( "Nenhum arquivo foi selecionado!", "Atenção" )
   ELSE
      xIMAGE( cArquivo )
   ENDIF

   DirChange( cDirExe ) // NO FUNCIONA

RETURN NIL

// FIN / END
