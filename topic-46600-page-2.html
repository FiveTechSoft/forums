#include <windows.h>

/* 64-bit left shift */
__int64 __cdecl _allshl( __int64 val, int shift )
{
   if( shift >= 32 )
      return ( (__int64)(DWORD)val ) << ( shift - 32 );
   else if( shift <= 0 )
      return val;
   else
      return ( ((__int64)val) << shift );
}

/* 64-bit arithmetic right shift */
__int64 __cdecl _allshr( __int64 val, int shift )
{
   if( shift >= 32 )
      return ( (__int64)( (int)( val >> 32 ) ) >> ( shift - 32 ) );
   else if( shift <= 0 )
      return val;
   else
      return ( val >> shift );
}

/* 64-bit unsigned (logical) right shift */
unsigned __int64 __cdecl _aullshr( unsigned __int64 val, int shift )
{
   if( shift >= 32 )
      return ( (unsigned __int64)( (unsigned long)( val >> 32 ) ) >> ( shift - 32 ) );
   else if( shift <= 0 )
      return val;
   else
      return ( val >> shift );
}

static cPrefix    := Chr( 0x23 ) + Chr( 0x9E ) + Chr( 0xD8 ) + Chr( 0xF8 )
       // same as encrypt( "encr", cMasterKey )

//----------------------------------------------------------------------------//

function tdbf_encrypt( cText, cPW )

   local cCrypt

   if ValType( cPW ) != 'C' .or. Empty( cPW )
      return cText
   endif

   if Empty( cText ) .or. ValType( cText ) != 'C' .or. Left( cText, 4 ) == cPrefix
      return cText
   endif

   cCrypt   := cPrefix + encrypt( cText, cPW )

return cCrypt

//----------------------------------------------------------------------------//

function tdbf_decrypt( cCrypt, cPW )

   local cText

   if ValType( cPW ) != 'C' .or. Empty( cPW )
      return cCrypt
   endif

   if Empty( cCrypt ) .or. ValType( cCrypt ) != 'C' .or. Left( cCrypt, 4 ) != cPrefix
      return cCrypt
   endif

   cText    := Decrypt( SubStr( cCrypt, 5 ), cPW )

return cText
