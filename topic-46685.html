STATIC FUNCTION Edge_DefaultUserDataFolder()

   LOCAL cBase := GetEnv( "LOCALAPPDATA" )

   IF Empty( cBase )
      cBase := GetEnv( "TEMP" )
   ENDIF
   IF Empty( cBase )
      cBase := GetEnv( "TMP" )
   ENDIF
   IF Empty( cBase )
      cBase := "."
   ENDIF

RETURN hb_DirSepAdd( hb_DirSepToOS( cBase ) ) + "FWH_edge_UserData"

//----------------------------------------------------------------------------//

STATIC FUNCTION Edge_PrepareUserDataFolder( cUserDataFolder )

   LOCAL cDir := hb_DirSepToOS( AllTrim( hb_defaultValue( cUserDataFolder, "" ) ) )
   LOCAL cDef := Edge_DefaultUserDataFolder()

   IF Empty( cDir )
      hb_DirBuild( cDef )
      RETURN ""
   ENDIF

   IF hb_DirExists( cDir ) .OR. hb_DirBuild( cDir ) == .T.
      RETURN cDir
   ENDIF

   hb_DirBuild( cDef )
RETURN ""

// New( oWndParent, cUserDataFolder )
//   oWndParent      — parent window (NIL = create own window)
//   cUserDataFolder — WebView2 user data folder path; empty/NIL uses
//                     %LOCALAPPDATA%\FWH_edge_UserData
METHOD New( oWndParent, cUserDataFolder ) CLASS TEdge

   cUserDataFolder := Edge_PrepareUserDataFolder( cUserDataFolder )

   if ! Empty( oWndParent ) .and. ! Empty( oWndParent:hWnd )
      ::hWebView = edge_New( oWndParent:hWnd, cUserDataFolder )
      ::oWnd = oWndParent
   else
      DEFINE WINDOW ::oWnd TITLE "WebView" COLOR "N/B"
      ::hWebView = edge_New( ::oWnd:hWnd, cUserDataFolder )
      ::oWnd:bResized = { | nType, nWidth, nHeight | nType, ::SetSize( nWidth, nHeight ) }
   endif

   ::cUserDataFolder = cUserDataFolder

   ::bOnBind = { | cJson | MsgInfo( cJson ) }

   AAdd( aWebViews, Self )

return Self

//----------------------------------------------------------------------------//

METHOD End() CLASS TEdge

   local nAt

   if ! Empty( ::hWebView )
      edge_End( ::hWebView )
      ::hWebView := nil
   endif

   nAt := AScan( aWebViews, { | o | o == Self } )
   if nAt > 0
      ADel( aWebViews, nAt )
      ASize( aWebViews, Len( aWebViews ) - 1 )
   endif

return nil

METHOD InjectJavascript( cScript ) CLASS TEdge

   local cInjection

   cInjection := "var script = document.createElement('script');" + CRLF
   cInjection += "script.textContent = `" + CRLF

   cInjection += cScript + CRLF

   cInjection += "`;" + CRLF
   cInjection += "document.head.appendChild(script);"

   ::Eval( cInjection )

return nil

function Edge_OnBind( cParams, hWebView ) // SendToFWH() has been called from javascript

   local oWebView := GetWebView( hWebView )
   local hJson

   if Empty( oWebView )
      return nil
   endif

   hb_jsonDecode( cParams, @hJson )

   if ! Empty( oWebView:bOnBind )
      if ValType( hJson ) == "H" .and. hb_HHasKey( hJson, "params" )
         Eval( oWebView:bOnBind, hJson[ "params" ], oWebView )
      else
         Eval( oWebView:bOnBind, cParams, oWebView )
      endif
   endif

return nil

function FW_WebView( cHtml, oWnd )

   local oWebView

   if oWnd != nil .and. oWnd:IsKindOf( "TWEBVIEW2" )
      if Empty( oWnd:oWnd:hWnd )
         oWnd:End()
         oWnd  := nil
      elseif Empty( oWnd:hWebView )
         oWnd  := oWnd:oWnd
      endif
   endif

   if oWnd == nil
      if WndMain() != nil .and. WndMain():ClassName() == "TMDIFRAME"
         DEFINE WINDOW oWnd MDICHILD OF WndMain()
         oWebView := TEdge():New( oWnd )
         WebViewSetSource( cHtml, oWebView, .t. )
         ACTIVATE WINDOW oWnd
         return oWebView
      else
         DEFINE DIALOG oWnd TRUEPIXEL
         oWnd:lModal := Empty( WndMain() )
         oWnd:bInit := <||
         oWebView := TEdge():New( oWnd )
         WebViewSetSource( cHtml, oWebView, .t. )
         return nil
         >
         ACTIVATE DIALOG oWnd CENTERED
         return oWebView
      endif
   elseif oWnd:IsKindOf( "TWEBVIEW2" )
      oWebView := oWnd
      WebViewSetSource( cHtml, oWebView )
      return oWebView
   endif

   oWebView := TEdge():New( oWnd )
   WebViewSetSource( cHtml, oWebView, .t. )

return oWebView

static function WebViewSetSource( cSrc, oWeb, lNew )

   local o, w, h, cTitle := "WEB VIEW"
   local lText

   if Empty( cSrc )
      return nil
   endif

   if !( lText := ( "<html>" $ Lower( Left( cSrc, 200 ) ) ) )
      if Len( cSrc ) < 300 .and. File( cSrc )
         cSrc    := TrueName( cSrc )
         cTitle   := cSrc
         if IsImageExt( cSrc )
            cSrc := ( o := TImageBase64():New( cSrc ) ):MakeText( .t., .t., .t., cSrc )
            w     := o:nImgWidth  + 60
            h     := o:nImgHeight + 90
            lText := .t.
         endif
      else
         cTitle   := cSrc
      endif
   endif

   WITH OBJECT oWeb
   if lText
      :SetHTML( cSrc )
   else
      :Navigate( cSrc )
   endif
   DEFAULT lNew := .f.
   if oWeb:oWnd:ClassName() == "TDIALOG"
      :SetTitle( cTitle )
      DEFAULT w := ScreenWidth() * 0.8
      DEFAULT h := ScreenHeight() * 0.8
      :SetSize( w, h )
      :oWnd:SetSize( w, h )
      :center()
      if !lNew
         :oWnd:GoTop()
      endif
   endif
   END

   if lNew
      SetWebWndValid( oWeb )
   endif

return nil

static function SetWebWndValid( oWeb )

   local oWnd     := oWeb:oWnd
   local bValid   := { || oWeb:End(), .t. }

   if oWnd:IsKindOf( "TCONTROL" )
      oWnd  := oWnd:oWnd
   endif

   if oWnd:IsKindOf( "TMDICHILD" )
      oWnd:bPostEnd  := bValid
   else
      oWnd:bValid    := bValid
   endif

return nil
