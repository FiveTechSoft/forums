Estimado Armando,

Code: Select all Collapse
#include "FiveWin.ch"  
  

#define ID_SAY_NAME   101  

#define ID_GET_NAME   102  

#define ID_SAY_CITY   103  

#define ID_GET_CITY   104  

#define ID_BTN_OK     105  

#define ID_BTN_EXIT   106  
  

FUNCTION Main()  
  

   LOCAL oDlg, oFont  

   LOCAL oSayName, oSayCity, oGetName, oGetCity, oBtnOk, oBtnExit  

   LOCAL nDpi := GetCurrentDpi()  

   LOCAL nScale := nDpi / 96.0  
  

   DEFINE FONT oFont NAME "Tahoma" SIZE 0, -Int( 9 * nScale )  
  

   DEFINE DIALOG oDlg RESOURCE "DLG_MAIN" PIXEL FONT oFont ;  

      TITLE "Autoajuste desde RC"  
  

   REDEFINE SAY      oSayName ID ID_SAY_NAME OF oDlg  

   REDEFINE GET      oGetName VAR cName ID ID_GET_NAME OF oDlg  

   REDEFINE SAY      oSayCity ID ID_SAY_CITY OF oDlg  

   REDEFINE GET      oGetCity  VAR cCity ID ID_GET_CITY OF oDlg  

   REDEFINE BUTTON   oBtnOk   ID ID_BTN_OK   OF oDlg ACTION oDlg:End()  

   REDEFINE BUTTON   oBtnExit  ID ID_BTN_EXIT OF oDlg ACTION oDlg:End()  
  

   oDlg:bResized := { || ResizeDlg( oDlg, oSayName, oGetName, oSayCity, oGetCity, oBtnOk, oBtnExit ) }  
  

   ACTIVATE DIALOG oDlg CENTERED ;  

      ON INIT ( oDlg:SetSize( Int( 300 * nScale ), Int( 180 * nScale ) ), oDlg:Center() )  
  

   RELEASE FONT oFont  
  

RETURN NIL  
  

STATIC FUNCTION ResizeDlg( oDlg, oSayName, oGetName, oSayCity, oGetCity, oBtnOk, oBtnExit )  
  

   LOCAL oRect := oDlg:GetCliRect()  

   LOCAL nW := oRect:nWidth  

   LOCAL nH := oRect:nHeight  
  

   oSayName:nTop := 14  

   oSayName:nLeft := 12  

   oGetName:nTop := 12  

   oGetName:nLeft := 68  

   oGetName:nWidth := nW - 80  
  

   oSayCity:nTop := 40  

   oSayCity:nLeft := 12  

   oGetCity:nTop := 38  

   oGetCity:nLeft := 68  

   oGetCity:nWidth := nW - 80  
  

   oBtnOk:nTop := nH - 28  

   oBtnExit:nTop := nH - 28  

   oBtnOk:nLeft := nW - 160  

   oBtnExit:nLeft := nW - 80  
  

   oGetName:Refresh()  

   oGetCity:Refresh()  

   oBtnOk:Refresh()  

   oBtnExit:Refresh()  
  

RETURN NIL  
  

FUNCTION GetCurrentDpi()  
  

   LOCAL hDc  := GetDc( 0 )  

   LOCAL nDpi := GetDeviceCaps( hDc, LOGPIXELSX )  
  

   ReleaseDC( 0, hDc )  
  

RETURN nDpi

resdlg.prg

Code: Select all Collapse
#include "FiveWin.ch"

#define LOGPIXELSX    88

#define ID_SAY_NAME   101
#define ID_GET_NAME   102
#define ID_SAY_CITY   103
#define ID_GET_CITY   104
#define ID_BTN_OK     105
#define ID_BTN_EXIT   106

#define DLG_ORIG_W    300
#define DLG_ORIG_H    180
#define FONT_BASE_SIZE 9

FUNCTION Main()

   LOCAL oDlg, oFont
   LOCAL oSayName, oSayCity, oGetName, oGetCity, oBtnOk, oBtnExit
   LOCAL cName := Space( 40 ), cCity := Space( 40 )
   LOCAL nDpi := GetCurrentDpi()
   LOCAL nScale := nDpi / 96.0

   DEFINE FONT oFont NAME "Tahoma" SIZE 0, -Int( FONT_BASE_SIZE * nScale )

   DEFINE DIALOG oDlg RESOURCE "DLG_MAIN" PIXEL FONT oFont ;
      TITLE "Autoajuste desde RC"

   REDEFINE SAY      oSayName ID ID_SAY_NAME OF oDlg
   REDEFINE GET      oGetName VAR cName ID ID_GET_NAME OF oDlg
   REDEFINE SAY      oSayCity ID ID_SAY_CITY OF oDlg
   REDEFINE GET      oGetCity  VAR cCity ID ID_GET_CITY OF oDlg
   REDEFINE BUTTON   oBtnOk   ID ID_BTN_OK   OF oDlg ACTION oDlg:End()
   REDEFINE BUTTON   oBtnExit  ID ID_BTN_EXIT OF oDlg ACTION oDlg:End()

   oDlg:bResized := { || ResizeDlg( oDlg, oSayName, oGetName, ;
                                     oSayCity, oGetCity, oBtnOk, oBtnExit ) }

   ACTIVATE DIALOG oDlg CENTERED ;
      ON INIT ( oDlg:SetSize( Int( DLG_ORIG_W * nScale ), Int( DLG_ORIG_H * nScale ) ), oDlg:Center() )

   RELEASE FONT oFont

RETURN NIL

STATIC FUNCTION ResizeDlg( oDlg, oSayName, oGetName, oSayCity, oGetCity, oBtnOk, oBtnExit )

   LOCAL oRect := oDlg:GetCliRect()
   LOCAL nW := oRect:nWidth
   LOCAL nH := oRect:nHeight
   LOCAL nSX := nW / DLG_ORIG_W
   LOCAL nSY := nH / DLG_ORIG_H
   LOCAL nFontSize := Int( FONT_BASE_SIZE * nSY )
   LOCAL oNewFont

   oSayName:nTop   := Int( 14 * nSY )
   oSayName:nLeft  := Int( 12 * nSX )
   oSayName:nWidth := Int( 50 * nSX )
   oSayName:nHeight := Int( 10 * nSY )

   oGetName:nTop    := Int( 12 * nSY )
   oGetName:nLeft   := Int( 68 * nSX )
   oGetName:nWidth  := Int( 220 * nSX )
   oGetName:nHeight := Int( 14 * nSY )

   oSayCity:nTop   := Int( 40 * nSY )
   oSayCity:nLeft  := Int( 12 * nSX )
   oSayCity:nWidth := Int( 50 * nSX )
   oSayCity:nHeight := Int( 10 * nSY )

   oGetCity:nTop    := Int( 38 * nSY )
   oGetCity:nLeft   := Int( 68 * nSX )
   oGetCity:nWidth  := Int( 220 * nSX )
   oGetCity:nHeight := Int( 14 * nSY )

   oBtnOk:nTop    := Int( 150 * nSY )
   oBtnOk:nLeft   := Int( 120 * nSX )
   oBtnOk:nWidth  := Int( 70 * nSX )
   oBtnOk:nHeight := Int( 16 * nSY )

   oBtnExit:nTop    := Int( 150 * nSY )
   oBtnExit:nLeft   := Int( 200 * nSX )
   oBtnExit:nWidth  := Int( 70 * nSX )
   oBtnExit:nHeight := Int( 16 * nSY )

   DEFINE FONT oNewFont NAME "Tahoma" SIZE 0, -nFontSize

   oDlg:SetFont( oNewFont )

   oSayName:Refresh()
   oGetName:Refresh()
   oSayCity:Refresh()
   oGetCity:Refresh()
   oBtnOk:Refresh()
   oBtnExit:Refresh()

RETURN NIL

FUNCTION GetCurrentDpi()

   LOCAL hDc  := GetDc( 0 )
   LOCAL nDpi := GetDeviceCaps( hDc, LOGPIXELSX )

   ReleaseDC( 0, hDc )

RETURN nDpi
